#    python -m venv venv

#    venv\Scripts\activate      (Windows)
#
# 2. Install dependencies:
#    pip install -r requirements.txt
#
# 3. YOLO model will auto-download on first run (yolov8n.pt ~6MB)
#
# 4. MySQL Setup:
#    CREATE DATABASE toll_monitoring CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
#    Then update DB_CONFIG in ai_engine.py
#
# 5. Run AI Engine:
#    python ai_engine.py --source 0           (webcam)
#    python ai_engine.py --source video.mp4   (video file)
#    python ai_engine.py --source rtsp://ip   (IP camera)
#
